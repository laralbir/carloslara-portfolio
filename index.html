// AWS SNS Fan-out & SQS Idempotent Consumer Pipeline
import { SNSClient, PublishCommand } from '@aws-sdk/client-sns';
import { SQSHandler, SQSEvent } from 'aws-lambda';
import { IdempotencyStore } from './infrastructure/idempotency';

export const publishInventoryDelta = async (payload: InventoryUpdate): Promise<void> => {
  const sns = new SNSClient({ region: 'eu-west-1' });
  await sns.send(new PublishCommand({
    TopicArn: process.env.INVENTORY_FANOUT_TOPIC_ARN,
    Message: JSON.stringify(payload),
    MessageAttributes: {
      channel: { DataType: 'String', StringValue: payload.channelId },
      traceId: { DataType: 'String', StringValue: payload.traceId }
    }
  }));
};

export const handleSqsMessage: SQSHandler = async (event: SQSEvent) => {
  for (const record of event.Records) {
    const { messageId, body } = record;
    if (await IdempotencyStore.hasProcessed(messageId)) continue;
    
    await processOtaSync(JSON.parse(body));
    await IdempotencyStore.markCompleted(messageId, { ttl: 86400 });
  }
};

import { McpServer } from '@modelcontextprotocol/sdk/server/mcp.js';
import { z } from 'zod';
import { ReservationQueryPort } from './ports/reservation-query.port';

const server = new McpServer({ name: 'avantio-prod-diagnostics', version: '2.4.0' });

// Diagnostic Tool: Inspect booking timeline across MongoDB & SQL Server
server.tool(
  'investigate_booking_anomaly',
  { bookingId: z.string(), otaChannel: z.enum(['booking', 'airbnb', 'expedia']) },
  async ({ bookingId, otaChannel }) => {
    const timeline = await ReservationQueryPort.reconstructAuditTrail(bookingId, otaChannel);

      content: [{ type: 'text', text: JSON.stringify(timeline, null, 2) }]
    };
  }
